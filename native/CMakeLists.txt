cmake_minimum_required(VERSION 3.20)
project(subgen_whisper_worker LANGUAGES CXX)
# Single-configuration generators must not silently ship Debug CRT dependencies.
if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
find_package(Vulkan REQUIRED)
set(WHISPER_CPP_SOURCE "" CACHE PATH "Pinned whisper.cpp source directory")
set(WHISPER_CPP_BUILD "" CACHE PATH "Matching prebuilt whisper.cpp directory")
foreach(header include/whisper.h ggml/include/ggml-backend.h examples/json.hpp)
  if(NOT EXISTS "${WHISPER_CPP_SOURCE}/${header}")
    message(FATAL_ERROR "Missing pinned whisper.cpp header: ${header}")
  endif()
endforeach()
find_library(WHISPER_LIB NAMES whisper PATHS "${WHISPER_CPP_BUILD}/bin" "${WHISPER_CPP_BUILD}/src" NO_DEFAULT_PATH REQUIRED)
find_library(GGML_LIB NAMES ggml PATHS "${WHISPER_CPP_BUILD}/bin" "${WHISPER_CPP_BUILD}/ggml/src" NO_DEFAULT_PATH REQUIRED)
find_library(GGML_BASE_LIB NAMES ggml-base PATHS "${WHISPER_CPP_BUILD}/bin" "${WHISPER_CPP_BUILD}/ggml/src" NO_DEFAULT_PATH REQUIRED)
add_executable(subgen-whisper-worker whisper_cpp_worker.cpp)
target_compile_features(subgen-whisper-worker PRIVATE cxx_std_17)
target_include_directories(subgen-whisper-worker PRIVATE
  "${WHISPER_CPP_SOURCE}/include" "${WHISPER_CPP_SOURCE}/ggml/include"
  "${WHISPER_CPP_SOURCE}/examples")
target_link_libraries(subgen-whisper-worker PRIVATE "${WHISPER_LIB}" "${GGML_LIB}" "${GGML_BASE_LIB}" Vulkan::Vulkan ${CMAKE_DL_LIBS})
add_executable(subgen-vulkan-probe vulkan_probe.cpp)
target_compile_features(subgen-vulkan-probe PRIVATE cxx_std_17)
target_include_directories(subgen-vulkan-probe PRIVATE "${WHISPER_CPP_SOURCE}/examples")
target_link_libraries(subgen-vulkan-probe PRIVATE Vulkan::Vulkan)
if(MSVC)
  target_compile_options(subgen-whisper-worker PRIVATE /utf-8 /W4)
  target_compile_options(subgen-vulkan-probe PRIVATE /utf-8 /W4)
else()
  target_compile_options(subgen-whisper-worker PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(subgen-vulkan-probe PRIVATE -Wall -Wextra -Wpedantic)
endif()
